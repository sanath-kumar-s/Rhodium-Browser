from PyQt6.QtCore import Qt, QObject
from PyQt6.QtGui import QAction, QKeySequence, QShortcut
from PyQt6.QtWebEngineCore import QWebEngineProfile, QWebEngineSettings
from PyQt6.QtWebEngineWidgets import QWebEngineView


class FullscreenManager(QObject):
    """
    Handles two kinds of fullscreen:
      1. Browser/window fullscreen (F11 or Ctrl+Shift+F). Hides browser chrome.
      2. HTML5 fullscreen requested by a web page (YouTube, Vimeo, etc.).

    Both can be active at once. The window stays fullscreen until both are off,
    and chrome is restored only when the window leaves fullscreen.
    """

    def __init__(self, window):
        super().__init__(window)
        self.window = window
        self.tab_widget = window.tab_widget

        self._user_fullscreen = False     # F11 / Ctrl+Shift+F
        self._html_fullscreen = False     # requested by a web page
        self._html_page = None            # page that currently owns HTML fullscreen
        self._applied = False             # True while the window is actually fullscreen

        self._saved_window_state = None
        self._saved_geometry = None
        self._saved_margins = None
        self._saved_visibility = {}

        self._setup_profile()
        self._setup_shortcuts()

        # Attach to views on startup and whenever the user switches tabs
        # (covers tabs created after startup).
        self.tab_widget.currentChanged.connect(lambda _index: self.attach_all())
        self.attach_all()

    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------

    def _setup_profile(self):
        settings = QWebEngineProfile.defaultProfile().settings()
        settings.setAttribute(QWebEngineSettings.WebAttribute.LocalStorageEnabled, True)
        settings.setAttribute(QWebEngineSettings.WebAttribute.FullScreenSupportEnabled, True)

    def _setup_shortcuts(self):
        self.f11_action = QAction("Toggle Fullscreen", self.window)
        self.f11_action.setShortcut(QKeySequence(Qt.Key.Key_F11))
        self.f11_action.triggered.connect(self.toggle_fullscreen)
        self.window.addAction(self.f11_action)

        self.ctrl_shift_f_action = QAction("Toggle Fullscreen (Alt)", self.window)
        self.ctrl_shift_f_action.setShortcut(QKeySequence("Ctrl+Shift+F"))
        self.ctrl_shift_f_action.triggered.connect(self.toggle_fullscreen)
        self.window.addAction(self.ctrl_shift_f_action)

        self.escape_shortcut = QShortcut(QKeySequence(Qt.Key.Key_Escape), self.window)
        self.escape_shortcut.activated.connect(self.on_escape)

    def attach_all(self):
        """Connect HTML fullscreen handling to every web view in the window."""
        for view in self.window.findChildren(QWebEngineView):
            self.attach_view(view)

    def attach_view(self, view):
        if getattr(view, "_fullscreen_attached", False):
            return
        view._fullscreen_attached = True

        view.settings().setAttribute(
            QWebEngineSettings.WebAttribute.FullScreenSupportEnabled, True
        )

        page = view.page()
        page.fullScreenRequested.connect(
            lambda request, p=page: self._on_html_fullscreen(request, p)
        )

    # ------------------------------------------------------------------
    # Public API
    # ------------------------------------------------------------------

    def is_fullscreen(self):
        return self._user_fullscreen or self._html_fullscreen

    def toggle_fullscreen(self):
        if self.is_fullscreen():
            self._user_fullscreen = False
            if self._html_fullscreen:
                self._exit_html_fullscreen()
        else:
            self._user_fullscreen = True
        self._apply()

    def on_escape(self):
        if self._html_fullscreen:
            self._exit_html_fullscreen()
        elif self._user_fullscreen:
            self._user_fullscreen = False
        else:
            return
        self._apply()

    # ------------------------------------------------------------------
    # HTML5 fullscreen requests from web pages
    # ------------------------------------------------------------------

    def _on_html_fullscreen(self, request, page):
        try:
            request.accept()
        except RuntimeError:
            return

        if request.toggleOn():
            self._html_fullscreen = True
            self._html_page = page
        else:
            self._html_fullscreen = False
            self._html_page = None

        self._apply()

    def _exit_html_fullscreen(self):
        page = self._html_page
        self._html_fullscreen = False
        self._html_page = None

        if page is not None:
            try:
                # Asking the page to leave its own fullscreen element is
                # the reliable way to end HTML5 fullscreen from the app side.
                page.runJavaScript(
                    "if (document.fullscreenElement) document.exitFullscreen();"
                )
            except RuntimeError:
                pass

    # ------------------------------------------------------------------
    # Window state transitions
    # ------------------------------------------------------------------

    def _apply(self):
        want = self.is_fullscreen()

        if want and not self._applied:
            self._enter()
        elif not want and self._applied:
            self._exit()

    def _enter(self):
        w = self.window
        self._applied = True

        # Save everything needed to restore the window exactly.
        self._saved_window_state = w.windowState()
        self._saved_geometry = w.saveGeometry()

        layout = w.centralWidget().layout() if w.centralWidget() else None
        self._saved_margins = layout.contentsMargins() if layout else None
        if layout:
            layout.setContentsMargins(0, 0, 0, 0)

        # Hide chrome. The tab bar is hidden, not the tab widget, so the
        # web content stays visible.
        self._saved_visibility = {}
        for widget in self._chrome_widgets():
            self._saved_visibility[widget] = not widget.isHidden()
            widget.hide()

        w.showFullScreen()

        view = self._current_view()
        if view is not None:
            view.setFocus(Qt.FocusReason.OtherFocusReason)

    def _exit(self):
        w = self.window
        self._applied = False

        # Restore chrome exactly as it was.
        for widget, was_visible in self._saved_visibility.items():
            widget.setVisible(was_visible)
        self._saved_visibility = {}

        layout = w.centralWidget().layout() if w.centralWidget() else None
        if layout and self._saved_margins is not None:
            layout.setContentsMargins(self._saved_margins)

        # Restore the previous window state.
        state = self._saved_window_state or Qt.WindowState.WindowNoState
        if state & Qt.WindowState.WindowMaximized:
            w.showMaximized()
        else:
            if self._saved_geometry is not None:
                w.restoreGeometry(self._saved_geometry)
            w.showNormal()

        self._saved_window_state = None
        self._saved_geometry = None
        self._saved_margins = None

        view = self._current_view()
        if view is not None:
            view.setFocus(Qt.FocusReason.OtherFocusReason)

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------

    def _chrome_widgets(self):
        w = self.window
        items = [
            getattr(w, "toolbar", None),
            getattr(w, "find_bar", None),
            self.tab_widget.tabBar(),
            w.statusBar(),
        ]
        return [item for item in items if item is not None]

    def _current_view(self):
        current = self.tab_widget.currentWidget()
        if current is None:
            return None
        if isinstance(current, QWebEngineView):
            return current
        return current.findChild(QWebEngineView)
